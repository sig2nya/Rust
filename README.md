```
int* r;
{
    int x = 5;
    r = &x;   // 스택 프레임 사라지는데 포인터 저장
}
printf("%d", *r); // 💣 런타임에서 터질 수도 있고, 아닐 수도 있고 — 완전 운빨
```
```
fn main() {
    let r;
    {
        let x = 5;
        r = &x; // ❌ 수명(lifetime) 위반 — 컴파일러가 즉시 차단
    }
    println!("{}", r);
}
```
